cmake_minimum_required(VERSION 3.10)
project(sport_motion_recorder CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
if(NOT CMAKE_BUILD_TYPE)
  set(CMAKE_BUILD_TYPE Release)
endif()
# unitree_sdk2 installed in /usr/local (go2-policies README 2-3)
find_package(unitree_sdk2 REQUIRED)
add_executable(record_sport_motion record_sport_motion.cpp)
target_link_libraries(record_sport_motion unitree_sdk2 pthread)
